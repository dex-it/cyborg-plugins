    process.stdout.write("arecord: \u043C\u0438\u043A\u0440\u043E\u0444\u043E\u043D Cyborg \u0434\u043B\u044F \u0443\u0434\u0430\u043B\u0451\u043D\u043D\u043E\u0439 \u0441\u0435\u0441\u0441\u0438\u0438\n");
    return 0;
  }
  const target = args[args.length - 1] ?? "-";
  const probe = target === "/dev/null";
  const refuse = (reason) => {
    const real = realArecord(env.PATH ?? "", own);
    if (real) return passThrough(real, args);
    process.stderr.write(`arecord: ${reason}
`);
    return Promise.resolve(1);
  };
  const sock2 = env.CYBORG_MIC;
  if (!sock2) return refuse("\u0441\u043E\u043A\u0435\u0442 \u043C\u0438\u043A\u0440\u043E\u0444\u043E\u043D\u0430 Cyborg \u043D\u0435 \u0437\u0430\u0434\u0430\u043D");
  const socket = import_node_net.default.connect(sock2);
  try {
    await new Promise((resolve, reject) => {
      socket.once("connect", resolve);
      socket.once("error", reject);
    });
  } catch {
    return refuse("\u0434\u0435\u043C\u043E\u043D Cyborg \u043D\u0435 \u043F\u0440\u0438\u043D\u0438\u043C\u0430\u0435\u0442 \u0437\u0430\u043F\u0438\u0441\u044C");
  }
  socket.write(probe ? "probe\n" : "record\n");
  let head;
  try {
    head = await readLine(socket, 3e4);
  } catch (error) {
    socket.destroy();
    return refuse(`\u043C\u0438\u043A\u0440\u043E\u0444\u043E\u043D Cyborg \u043D\u0435 \u043E\u0442\u0432\u0435\u0442\u0438\u043B: ${error.message}`);
  }
  if (head.line !== "ok") {
    socket.destroy();
    return refuse(head.line.replace(/^no /, "") || "\u043C\u0438\u043A\u0440\u043E\u0444\u043E\u043D Cyborg \u043D\u0435\u0434\u043E\u0441\u0442\u0443\u043F\u0435\u043D");
  }
  if (probe) {
    socket.destroy();
    return 0;
  }
  const out = target === "-" ? process.stdout : import_node_fs5.default.createWriteStream(target);
  return new Promise((resolve) => {
    let done = false;
    const finish = (code, flush = true) => {
      if (done) return;
      done = true;
      socket.destroy();
      if (flush && out !== process.stdout) out.end(() => resolve(code));
      else resolve(code);
    };
    process.once("SIGTERM", () => finish(0));
    process.once("SIGINT", () => finish(0));
    out.on("error", () => finish(0, false));
    if (head.rest.length) out.write(head.rest);
    socket.pipe(out);
    socket.on("end", () => finish(0));
    socket.on("error", () => finish(1));
  });
}
function passThrough(file, args) {
  return new Promise((resolve) => {
    const child = (0, import_node_child_process3.spawn)(file, args, { stdio: "inherit" });
    const stop = () => void child.kill("SIGTERM");
    process.once("SIGTERM", stop);
    process.once("SIGINT", stop);
    child.on("error", () => resolve(1));
    child.on("close", (code) => resolve(code ?? 0));
  });
}
function arecordScript(node, script, quote) {
  return `#!/bin/sh

`;
}
